    <script id="worker" type="javascript/worker">
        self.importScripts('https://cdn.jsdelivr.net/npm/occt-import-js@0.0.23/dist/occt-import-js.min.js');

        self.onmessage = async (e) => {
            const { fileBuffer } = e.data;
            try {
                const occt = await occtimportjs({
                    locateFile: (name) => `https://cdn.jsdelivr.net/npm/occt-import-js@0.0.23/dist/${name}`
                });
                const result = occt.ReadStepFile(fileBuffer, null);
                self.postMessage({ success: true, result });
            } catch (err) {
                self.postMessage({ success: false, error: err.message });
            }
        };
    </script>

    <script>
        document.addEventListener('DOMContentLoaded', () => {
            // DOM 元素獲取
            const container = document.getElementById('viewer-container');
            const canvas = document.getElementById('viewer-canvas');
            const loadingIndicator = document.getElementById('loading-indicator');
            const fileInput = document.getElementById('file-input');
            const infoDisplay = document.getElementById('info-display');
            const bgColorPicker = document.getElementById('bg-color-picker');
            const modelColorPicker = document.getElementById('model-color-picker');
            const edgeColorPicker = document.getElementById('edge-color-picker');
            const toggleEdgesBtn = document.getElementById('toggle-edges-btn');
            const explodeSlider = document.getElementById('explode-slider');
            const resetViewBtn = document.getElementById('reset-view-btn');
            const frontViewBtn = document.getElementById('front-view-btn');
            const topViewBtn = document.getElementById('top-view-btn');
            const sideViewBtn = document.getElementById('side-view-btn');
            const isoViewBtn = document.getElementById('iso-view-btn');
            const measureBtn = document.getElementById('measure-btn');
            const clearMeasureBtn = document.getElementById('clear-measure-btn');
            const measurementDisplay = document.getElementById('measurement-display');


            let scene, camera, renderer, controls, mainObject, initialCameraState;
            let meshes = [], edgeLines = [], partOrigins = [];
            let isMeasuring = false, measurePoints = [];
            let measureLines = new THREE.Group();

            // Web Worker 初始化
            const workerBlob = new Blob([document.getElementById('worker').textContent], { type: 'application/javascript' });
            const worker = new Worker(URL.createObjectURL(workerBlob));
            
            worker.onmessage = (e) => {
                const { success, result, error } = e.data;
                loadingIndicator.style.display = 'none';
                if (success) {
                    buildSceneFromParsedResult(result);
                } else {
                    console.error('Worker Error:', error);
                    alert('在背景執行緒中解析檔案時發生錯誤。');
                }
            };

            function initViewer() {
                scene = new THREE.Scene();
                scene.background = new THREE.Color(0xffffff);

                camera = new THREE.PerspectiveCamera(45, container.clientWidth / container.clientHeight, 0.1, 1000);
                camera.position.set(200, 200, 200);

                renderer = new THREE.WebGLRenderer({ canvas: canvas, antialias: true });
                renderer.setPixelRatio(window.devicePixelRatio);
                renderer.setSize(container.clientWidth, container.clientHeight);
                // *** CHANGE: Removed clipping plane logic ***
                // renderer.localClippingEnabled = true; 

                controls = new THREE.OrbitControls(camera, renderer.domElement);
                controls.enableDamping = true;

                const ambientLight = new THREE.AmbientLight(0xffffff, 0.8);
                scene.add(ambientLight);
                const directionalLight = new THREE.DirectionalLight(0xffffff, 1.0);
                directionalLight.position.set(1, 1.5, 1).normalize();
                scene.add(directionalLight);

                mainObject = new THREE.Object3D();
                scene.add(mainObject);
                scene.add(measureLines);

                animate();
                setupEventListeners();
            }

            function animate() {
                requestAnimationFrame(animate);
                controls.update();
                renderer.render(scene, camera);
            }

            function loadFile(file) {
                const reader = new FileReader();
                reader.onload = (e) => {
                    loadingIndicator.style.display = 'block';
                    const fileBuffer = new Uint8Array(e.target.result);
                    worker.postMessage({ fileBuffer });
                    document.getElementById('info-filename').textContent = file.name;
                };
                reader.readAsArrayBuffer(file);
            }
            
            function buildSceneFromParsedResult(result) {
                while(mainObject.children.length > 0){ 
                    const child = mainObject.children[0];
                    mainObject.remove(child);
                    if (child.geometry) child.geometry.dispose();
                    if (child.material) {
                        if (Array.isArray(child.material)) {
                            child.material.forEach(m => m.dispose());
                        } else {
                            child.material.dispose();
                        }
                    }
                }
                measureLines.clear();
                meshes = [];
                edgeLines = [];
                partOrigins = [];
                
                if (!result || result.meshes.length === 0) {
                    alert('無法從 STEP 檔案中讀取有效的幾何體。');
                    return;
                }

                result.meshes.forEach(resultMesh => {
                    const geometry = new THREE.BufferGeometry();
                    geometry.setAttribute('position', new THREE.BufferAttribute(new Float32Array(resultMesh.attributes.position.array), 3));
                    if (resultMesh.attributes.normal?.array) {
                        geometry.setAttribute('normal', new THREE.BufferAttribute(new Float32Array(resultMesh.attributes.normal.array), 3));
                    } else {
                        geometry.computeVertexNormals();
                    }
                    geometry.setIndex(new THREE.BufferAttribute(new Uint32Array(resultMesh.index.array), 1));
                    
                    const material = new THREE.MeshStandardMaterial({
                        color: new THREE.Color(modelColorPicker.value),
                        metalness: 0.2,
                        roughness: 0.7,
                        // *** CHANGE: Removed clipping plane logic ***
                        // clippingPlanes: [clippingPlane],
                        // clipShadows: true
                    });
                    if (resultMesh.color) {
                        material.color.setRGB(resultMesh.color[0], resultMesh.color[1], resultMesh.color[2]);
                    }
                    const mesh = new THREE.Mesh(geometry, material);
                    meshes.push(mesh);
                    mainObject.add(mesh);
                    
                    const edges = new THREE.EdgesGeometry(geometry, 1);
                    const lineMaterial = new THREE.LineBasicMaterial({ color: new THREE.Color(edgeColorPicker.value) });
                    const lineSegment = new THREE.LineSegments(edges, lineMaterial);
                    edgeLines.push(lineSegment);
                    mainObject.add(lineSegment);
                });

                fitCameraToObject(mainObject, 1.5);
                updateModelInfo();
                
                explodeSlider.value = 0;
                if (!toggleEdgesBtn.classList.contains('active')) {
                    toggleEdgesBtn.classList.add('active');
                }
            }

            function fitCameraToObject(object, offset = 1.5) {
                const box = new THREE.Box3().setFromObject(object);

                if (box.isEmpty()) {
                    console.warn("Cannot fit camera to an empty object.");
                    return;
                }

                const center = box.getCenter(new THREE.Vector3());
                const size = box.getSize(new THREE.Vector3());
                const maxDim = Math.max(size.x, size.y, size.z);
                
                const cameraFar = maxDim * 5;
                camera.far = cameraFar;
                camera.near = cameraFar / 1000;
                camera.updateProjectionMatrix();

                const fov = camera.fov * (Math.PI / 180);
                const cameraZ = Math.abs(maxDim / 2 / Math.tan(fov / 2));
                
                const newPosition = new THREE.Vector3(center.x, center.y, center.z + cameraZ * offset);
                
                camera.position.copy(newPosition);
                controls.target.copy(center);
                
                initialCameraState = {
                    position: camera.position.clone(),
                    target: controls.target.clone()
                };
                
                controls.update();
            }
            
            function updateModelInfo() {
                const box = new THREE.Box3().setFromObject(mainObject);
                if (box.isEmpty()) {
                    infoDisplay.style.display = 'none';
                    return;
                }
                const size = box.getSize(new THREE.Vector3());
                let totalFaces = 0;
                meshes.forEach(m => {
                    totalFaces += m.geometry.index.count / 3;
                });
                document.getElementById('info-size').textContent = `${size.x.toFixed(2)} x ${size.y.toFixed(2)} x ${size.z.toFixed(2)}`;
                document.getElementById('info-faces').textContent = totalFaces.toLocaleString();
                infoDisplay.style.display = 'block';
            }

            function setupEventListeners() {
                fileInput.addEventListener('change', (e) => loadFile(e.target.files[0]));
                
                bgColorPicker.addEventListener('input', (e) => scene.background.set(e.target.value));
                modelColorPicker.addEventListener('input', (e) => meshes.forEach(m => m.material.color.set(e.target.value)));
                edgeColorPicker.addEventListener('input', (e) => edgeLines.forEach(l => l.material.color.set(e.target.value)));
                toggleEdgesBtn.addEventListener('click', () => {
                    const isActive = toggleEdgesBtn.classList.toggle('active');
                    edgeLines.forEach(l => l.visible = isActive);
                });

                resetViewBtn.addEventListener('click', () => {
                    if (initialCameraState) {
                        camera.position.copy(initialCameraState.position);
                        controls.target.copy(initialCameraState.target);
                    }
                });
                
                const setView = (x, y, z) => {
                    if (!initialCameraState) return;
                    const offset = camera.position.distanceTo(controls.target);
                    camera.position.copy(controls.target).add(new THREE.Vector3(x, y, z).setLength(offset));
                    camera.lookAt(controls.target);
                };
                frontViewBtn.addEventListener('click', () => setView(0, 0, 1));
                topViewBtn.addEventListener('click', () => setView(0, 1, 0));
                sideViewBtn.addEventListener('click', () => setView(1, 0, 0));
                isoViewBtn.addEventListener('click', () => setView(1, 1, 1));

                explodeSlider.addEventListener('input', (e) => {
                    const value = parseFloat(e.target.value);
                    document.getElementById('explode-value').textContent = value.toFixed(1);
                    if (partOrigins.length !== meshes.length) {
                        partOrigins = meshes.map(mesh => {
                            const center = new THREE.Vector3();
                            mesh.geometry.computeBoundingBox();
                            mesh.geometry.boundingBox.getCenter(center);
                            return center;
                        });
                    }
                    const modelCenter = new THREE.Box3().setFromObject(mainObject).getCenter(new THREE.Vector3());
                    meshes.forEach((mesh, i) => {
                        const direction = partOrigins[i].clone().sub(modelCenter).normalize();
                        mesh.position.copy(direction.multiplyScalar(value * 10));
                        edgeLines[i].position.copy(mesh.position);
                    });
                });
                
                measureBtn.addEventListener('click', () => {
                    isMeasuring = !isMeasuring;
                    measureBtn.classList.toggle('active', isMeasuring);
                    controls.enabled = !isMeasuring;
                });
                clearMeasureBtn.addEventListener('click', () => {
                    measureLines.clear();
                    measurementDisplay.style.display = 'none';
                });
                
                canvas.addEventListener('click', (event) => {
                    if (!isMeasuring) return;
                    const rect = renderer.domElement.getBoundingClientRect();
                    const mouse = new THREE.Vector2();
                    mouse.x = ((event.clientX - rect.left) / rect.width) * 2 - 1;
                    mouse.y = -((event.clientY - rect.top) / rect.height) * 2 + 1;
                    
                    const raycaster = new THREE.Raycaster();
                    raycaster.setFromCamera(mouse, camera);
                    const intersects = raycaster.intersectObjects(meshes);
                    
                    if (intersects.length > 0) {
                        const point = intersects[0].point;
                        measurePoints.push(point);
                        
                        const dotGeometry = new THREE.SphereGeometry(0.5, 16, 16);
                        const dotMaterial = new THREE.MeshBasicMaterial({ color: 0xff0000 });
                        const dot = new THREE.Mesh(dotGeometry, dotMaterial);
                        dot.position.copy(point);
                        measureLines.add(dot);
                        
                        if (measurePoints.length === 2) {
                            const lineGeometry = new THREE.BufferGeometry().setFromPoints(measurePoints);
                            const lineMaterial = new THREE.LineBasicMaterial({ color: 0xff0000, linewidth: 2 });
                            const line = new THREE.Line(lineGeometry, lineMaterial);
                            measureLines.add(line);
                            
                            const distance = measurePoints[0].distanceTo(measurePoints[1]);
                            measurementDisplay.textContent = `距離: ${distance.toFixed(2)} mm`;
                            measurementDisplay.style.display = 'block';
                            
                            measurePoints = [];
                            isMeasuring = false;
                            measureBtn.classList.remove('active');
                            controls.enabled = true;
                        }
                    }
                });
            }

            initViewer();
        });
    </script>
